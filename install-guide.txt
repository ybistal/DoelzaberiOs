eng:
on first boot from live CD
login - root
password - doelzaberi
once it has booted, type doelzaberi-install
doelzaberi-install --list # just show the list of disks
doelzaberi-install --dry-run /dev/nvme0n1 # show the plan without writing anything
doelzaberi-install --gui /dev/sda # go straight to GUI (niri) without asking
doelzaberi-install --no-gui /dev/sda # go straight to no-GUI
doelzaberi-install -y /dev/sda # without extra confirmations
doelzaberi-install --user dunya -y /dev/sdc # user dunya, don't ask
doelzaberi-install --root-size 40G /dev/sda
doelzaberi-install --no-nvram /dev/sda # don't write a boot entry to UEFI

--esp-size 512M size of the EFI partition
--root-size 40G size of the btrfs partition, the rest stays free
--source DIR where to take boot/bzImage and boot/initrd from (default: the medium you booted from)
--no-initrd don't install initrd; there will be no live RAM entry
--no-nvram don't write a boot entry to the firmware (useful for testing)
--force continue even if the disk is in use or it is the live medium itself
--no-user don't create an account at all

doelzaberi-gui
rc-service seatd start # device management (input/GPU)
rc-service dbus start # D-Bus (niri requires it — built with --features dbus)

doelzaberi-gui --compositor=niri # explicitly specify the compositor
doelzaberi-gui --compositor=weston # go back to weston
doelzaberi-gui -p
